export function getVar(token) {
  if (typeof document === 'undefined') return '';
  return getComputedStyle(document.documentElement).getPropertyValue(token).trim();
}

export function hexToRgb(hex) {
  const clean = hex.replace(/^#/, '');
  if (clean.length !== 6) return null;
  return {
    r: parseInt(clean.slice(0, 2), 16),
    g: parseInt(clean.slice(2, 4), 16),
    b: parseInt(clean.slice(4, 6), 16),
  };
}

export function relativeLuminance({ r, g, b }) {
  return [r, g, b]
    .map(c => { const s = c / 255; return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4); })
    .reduce((acc, c, i) => acc + c * [0.2126, 0.7152, 0.0722][i], 0);
}

export function contrastVsWhite(hex) {
  const rgb = hexToRgb(hex);
  if (!rgb) return null;
  const l = relativeLuminance(rgb);
  return (1.05) / (l + 0.05);
}

export const ColorSwatch = ({ token, description, showContrast }) => {
  const [value, setValue] = useState('');
  const [ratio, setRatio] = useState(null);

  useEffect(() => {
    const val = getVar(token);
    setValue(val);
    if (showContrast && val.startsWith('#')) {
      setRatio(contrastVsWhite(val));
    }
  }, [token, showContrast]);

  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 16, padding: '10px 0', borderBottom: '1px solid #f0f0f0' }}>
      <div style={{
        width: 48, height: 48, borderRadius: 8, flexShrink: 0,
        background: value || `var(${token})`,
        border: '1px solid rgba(0,0,0,0.1)',
      }} />
      <div style={{ flex: 1, minWidth: 0 }}>
        <code style={{ fontSize: 12, color: '#0f385a', display: 'block', marginBottom: 2 }}>{token}</code>
        {description && (
          <div style={{ fontSize: 11, color: '#627380', marginBottom: 3, lineHeight: 1.4 }}>{description}</div>
        )}
        <span style={{ fontSize: 11, fontFamily: 'monospace', color: '#3b4651', background: '#f5f5f5', padding: '1px 6px', borderRadius: 3 }}>
          {value || '—'}
        </span>
      </div>
      {showContrast && ratio !== null && (
        <div style={{
          fontSize: 11, padding: '3px 8px', borderRadius: 4, whiteSpace: 'nowrap', flexShrink: 0,
          background: ratio >= 4.5 ? '#f7fbec' : ratio >= 3 ? '#fff7e8' : '#fff6f9',
          color: ratio >= 4.5 ? '#55701b' : ratio >= 3 ? '#b44a09' : '#e0006e',
          border: `1px solid ${ratio >= 4.5 ? '#6f921e' : ratio >= 3 ? '#d96c06' : '#e0006e'}`,
        }}>
          {ratio.toFixed(2)}:1 {ratio >= 4.5 ? 'AA ✓' : ratio >= 3 ? 'AA Large' : '✗ Falla'}
        </div>
      )}
    </div>
  );
};

export const ColorGroup = ({ title, tokens, showContrast = false }) => (
  <div style={{ marginBottom: 40 }}>
    <h3 style={{ fontFamily: 'Poppins, sans-serif', fontSize: 13, fontWeight: 600, color: '#627380', textTransform: 'uppercase', letterSpacing: '0.08em', marginBottom: 4, paddingBottom: 8, borderBottom: '2px solid #e5e9ec' }}>
      {title}
    </h3>
    {tokens.map(([name, desc]) => (
      <ColorSwatch key={name} token={name} description={desc} showContrast={showContrast} />
    ))}
  </div>
);

# Color

Los tokens de color del Poli Design System están organizados por **rol semántico**, no por valor. Los valores se resuelven en **tiempo de ejecución** directamente desde las CSS custom properties, por lo que siempre reflejan la paleta compilada por Style Dictionary.
