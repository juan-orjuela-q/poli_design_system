export function getVar(token) {
  if (typeof document === 'undefined') return '';
  return getComputedStyle(document.documentElement).getPropertyValue(token).trim();
}

export const FontFamilyRow = ({ token, label }) => {
  const [value, setValue] = useState('');
  useEffect(() => { setValue(getVar(token)); }, [token]);
  return (
    <div style={{ display: 'flex', alignItems: 'baseline', gap: 24, padding: '12px 0', borderBottom: '1px solid #f0f0f0' }}>
      <div style={{ width: 220, flexShrink: 0 }}>
        <code style={{ fontSize: 12, color: '#0f385a' }}>{token}</code>
        <div style={{ fontSize: 11, color: '#627380', marginTop: 2 }}>{label}</div>
      </div>
      <div style={{ fontFamily: `var(${token}), sans-serif`, fontSize: 22, color: '#0c0f12' }}>
        La luz plateada de las estrellas rompe la noche
      </div>
      <code style={{ fontSize: 11, color: '#627380', marginLeft: 'auto' }}>{value}</code>
    </div>
  );
};

export const TypeScaleRow = ({ sizeToken, label, fontFamily = '--text-headings' }) => {
  const [size, setSize] = useState('');
  const [family, setFamily] = useState('Poppins');
  useEffect(() => {
    setSize(getVar(sizeToken));
    setFamily(getVar(fontFamily) || 'Poppins');
  }, [sizeToken, fontFamily]);
  return (
    <div style={{ display: 'flex', alignItems: 'baseline', gap: 16, padding: '10px 0', borderBottom: '1px solid #f0f0f0', flexWrap: 'wrap' }}>
      <div style={{ width: 200, flexShrink: 0 }}>
        <code style={{ fontSize: 11, color: '#0f385a', display: 'block' }}>{sizeToken}</code>
        <span style={{ fontSize: 11, color: '#627380' }}>{label}</span>
      </div>
      <div style={{ flex: 1, fontFamily: `${family}, sans-serif`, fontSize: `var(${sizeToken})`, lineHeight: 1.2, color: '#0c0f12', overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>
        El veloz zorro marrón salta sobre el perro perezoso
      </div>
      <code style={{ fontSize: 11, color: '#3b4651', background: '#f5f5f5', padding: '1px 6px', borderRadius: 3, flexShrink: 0 }}>{size}</code>
    </div>
  );
};

      color: '#0c0f12',
      marginBottom: 12,
      overflow: 'hidden',
      display: '-webkit-box',
      WebkitLineClamp: 2,
      WebkitBoxOrient: 'vertical',

    </div>
    <div style={{ display: 'flex', flexWrap: 'wrap', alignItems: 'center', gap: 6 }}>
      <strong style={{ fontSize: 12, color: '#0f385a', minWidth: 180, marginRight: 4 }}>{name}</strong>
      {[
        { label: 'Familia',     token: familyToken,     value: family },
        { label: 'Tamaño',      token: sizeToken,       value: size },
        { label: 'Peso',        token: weightToken,     value: String(weight) },
        { label: 'Line-height', token: lineHeightToken, value: String(lineHeight) },
        ...(letterSpacing > 0 ? [{ label: 'Tracking', token: `${letterSpacing}em`, value: '' }] : []),

          {value && <span style={{ fontSize: 10, color: '#3b4651' }}>{value}</span>}
        </div>
      ))}
    </div>
  </div>

# Typography

El sistema tipográfico del Poli Design System usa **Poppins** para títulos y componentes interactivos, y **Open Sans** para cuerpos de texto. Los valores se resuelven en tiempo de ejecución desde las CSS custom properties.

## Display

Estilos para textos de máximo impacto visual: heroes, portadas y landings. Poppins SemiBold con interlineado muy ajustado.

> **Tokens compuestos** — Los estilos tipográficos combinan múltiples tokens. Úsalos siempre con sus tokens individuales, no con valores sueltos:
> ```scss
> h1 {
>   font-family: var(--text-headings);
>   font-size: var(--font-size-f-4xl);
>   font-weight: var(--font-weight-w-semibold);
>   line-height: var(--line-height-lh-tight);
> }
> ```

{displayStyles.map(s => <TextStyleCard key={s.name} {...s} />)}

---

## Headings

Jerarquía editorial H1–H6. Poppins SemiBold. Los tokens de tamaño corresponden a la escala `f-*`. El interlineado se ajusta según el nivel: **tight (1.2)** para H1–H5 y **base (1.5)** para H6.

{headingStyles.map(s => <TextStyleCard key={s.name} {...s} />)}

---

## Body

Estilos de cuerpo de texto. Open Sans. Tres tamaños × variante Regular / Bold. El tracking (letter-spacing) de `0.02em` aparece solo en las variantes Bold.

{bodyStyles.map(s => <TextStyleCard key={s.name} {...s} />)}
